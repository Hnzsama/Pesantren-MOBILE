Table: schools
Structure:
  - ulid('id')->primary()
  - string('name')
  - string('slug')->unique(); // For identification e.g. 'center', 'branch-1'
  - string('address')->nullable()
  - string('phone')->nullable()
  - string('email')->nullable()
  - string('theme_color')->nullable(); // Hex color for theme customization
  - string('logo_path')->nullable(); // Path to school logo
  - timestamps()
